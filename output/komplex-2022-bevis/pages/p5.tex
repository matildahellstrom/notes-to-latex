\begin{theorem}[Cauchys integralformel version 1]
$f$ holo i $G$, $\overline{D}(a, R) \subseteq G$, då är
\[ f(a) = \frac{1}{2\pi i}\int_{|z - a| = r} \frac{f(z)}{z - a}\,dz, \qquad 0 < r \le R \]
\end{theorem}
\textcolor{magenta!80!black}{$\uparrow$ Bevisa denna}

\begin{note}
\underline{Cauchys integralformel version 2}

$f$ holo i $G$, $a \in G$, $\gamma \sim_{G \setminus \{a\}} C(a, r)$ för små $r$, då är
\[ f(a) = \frac{1}{2\pi i}\int_\gamma \frac{f(z)}{z - a}\,dz \]
(för bevis av CIF v2 använd CIF v1 + CS)
\end{note}
