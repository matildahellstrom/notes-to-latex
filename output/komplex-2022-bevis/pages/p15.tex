\begin{theorem}[Klassifikation av nollställen]
Om $f$ holo i $G$, $f(a) = 0$ då är antingen
\begin{enumerate}[label=\roman*)]
  \item $f(z) = (z - a)^m g(z)$, $m \ge 1$, $g$ holo i $G$, $g(a) \neq 0$

  eller
  \item $f = 0$ i $G$
\end{enumerate}
$m$ kallas ordningen (alt.\ multipliciteten) av nollstället\\
(ii: $m = \infty$)
\end{theorem}
