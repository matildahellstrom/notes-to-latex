\begin{theorem}[Taylorutveckling av holomorfa funktioner]
Antag $f$ holo i $D(a, R)$. Då har $f$ en Taylorutveckling
\[ f(z) = \sum_0^\infty c_k(z - a)^k \quad \text{i } D(a, R), \text{ där} \]
\[ c_k = \frac{f^{(k)}(a)}{k!} = \frac{1}{2\pi i}\int_{|z - a| = r} \frac{f(z)}{(z - a)^{k + 1}}\,dz, \qquad 0 < r < R \]
\end{theorem}
