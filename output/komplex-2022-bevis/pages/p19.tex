\begin{theorem}[Laurentserieutveckling av holo funktioner]
Antag $f$ holo i $A(a, R_1, R_2)$. Då kan $f$ skrivas som
\[ f(z) = \sum_{-\infty}^\infty c_k(z - a)^k \quad \text{i } A(a, R_1, R_2) \text{ där} \]
\[ c_k = \frac{1}{2\pi i}\int_{|z - a| = r} \frac{f(z)}{(z - a)^{k + 1}}\,dz, \qquad R_1 < r < R_2. \]
\end{theorem}

\noindent\rule{\linewidth}{0.4pt}

\underline{OBS!} \quad $\dfrac{1}{1 - z} \overset{|z| < 1}{=} \displaystyle\sum_0^\infty z^k$
