\begin{proof}
$h := f - g$ holo i $G$. $h(a_n) = 0$ $\forall n$,
$h(a) = \lim_{n \to \infty} h(a_n) = 0$. $a$ nollställe till $h$ men ej isolerat ty $a_n \to a \Rightarrow h = 0$ i $G$, dvs $f = g$ i $G$.

\begin{framed}
Om $f$ holo i $G$, $f(a) = 0$ men $f$ ej identiskt lika med $0$ i $G$, då är $a$ ett \rattat{isolerat}{isolelat} nollställe, dvs $\exists\, r > 0 : f(z) \neq 0$ i $D(a, r) \setminus \{a\}$
\end{framed}
\underline{Bevis av bubblan}\\
$f$ ej identisk $= 0 \underset{\text{klass.}}{\Rightarrow} f(z) = (z - a)^m g(z)$, $g$ holo i $G$, $g(a) \neq 0$. $g$ spec.\ kont.\\
$\Rightarrow \exists\, r > 0 : g(z) \neq 0$ i $D(a, r) \Rightarrow f(z) \neq 0$ i $D(a, r) \setminus \{a\}$
\end{proof}
