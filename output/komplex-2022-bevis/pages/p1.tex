\section{Bevislista}
\begin{theorem}[Satsen om Cauchy--Riemanns ekvationer]
Låt $G \subseteq \mathbb{C}$ öppen, $f : G \to \mathbb{C}$, $u(x, y) := \operatorname{Re} f(x + iy)$, $v(x, y) := \operatorname{Im} f(x + iy)$.
\begin{enumerate}[label=\alph*)]
  \item Om $f$ holomorf i $G$ så löser $u, v$ CR:s ekv:
  \[ \begin{cases} \dfrac{\partial u}{\partial x} = \dfrac{\partial v}{\partial y} \\[1em] \dfrac{\partial u}{\partial y} = -\dfrac{\partial v}{\partial x} \end{cases} \qquad \text{eller ekvivalent} \quad \frac{\partial f}{\partial x} = -i\frac{\partial f}{\partial y} \]
  speciellt gäller att $f' = \dfrac{\partial f}{\partial x} = -i\dfrac{\partial f}{\partial y}$
  \item Om $f$ är $C^1$ och $u, v$ löser CR:s ekvation då är $f$ holomorf
\end{enumerate}
\end{theorem}
