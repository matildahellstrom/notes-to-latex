\begin{theorem}[Klassifikation av isolerade singulariteter (sats 4.1, 4.2 R)]
Antag $a$ isolerad singularitet till $f$. Då är $a$
\begin{enumerate}[label=\alph*)]
  \item hävbar omm $\lim_{z \to a} (z - a)f(z) = 0$
  \item en pol omm $f(z) = \dfrac{g(z)}{(z - a)^m}$ i $D(a, R)^\times$, $m \ge 1$, $g$ holo i $D(a, R)$, $g(a) \neq 0$.

  $m$ kallas polens ordning/multiplicitet
\end{enumerate}
\end{theorem}
