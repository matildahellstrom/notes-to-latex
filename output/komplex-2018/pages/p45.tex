\begin{proof}
Antag $\exists w \in G : u(w) = \sup_{z \in G} u(z)$.

Vill visa $u \equiv u(w)$ i $G$.
\[ \left\{ \begin{array}{l} \textbf{Steg 1:} \text{ Visar att om } \overline{D}(w, R) \subseteq G \Rightarrow u \equiv u(w) \text{ i } \overline{D}(w, R) \\ \textbf{Steg 2:} \text{ Visar } u \equiv u(w) \text{ i hela } G. \end{array} \right. \]

\textcolor{red!70!black}{\textbf{Steg 1:}} Antag $\overline{D}(w, R) \subseteq G$. Välj $0 < r \le R$. Enligt medelvärdessatsen $u(w) = \dfrac{1}{2\pi} \displaystyle\int_0^{2\pi} u(w + re^{it})\,dt$. Om $u(w + re^{it}) < u(w)$ för $t_0 \in [0, 2\pi]$ då
\notefigure{fig1}{}
\[ \underset{\textcolor{magenta!80!black}{u \text{ kont.}}}{\Rightarrow} u(w + re^{it}) \le u(w) - \varepsilon \text{ för } t \in [a, b],\ b > a,\ \varepsilon > 0. \]
\[ \Rightarrow u(w) = \frac{1}{2\pi} \int_0^{2\pi} u(w + re^{it})\,dt \le \frac{1}{2\pi}\left( \int_a^b u(w) - \varepsilon\,dt + \underbrace{\int u(w)\,dt}_{\text{resten}} \right) = \]
\[ = u(w) - \frac{\varepsilon(b - a)}{2\pi} \quad \textcolor{red!70!black}{\underline{\text{Motsägelse!}}} \]
$\Rightarrow u \equiv u(w)$ i $\overline{D}(w, R)$.

\textcolor{red!70!black}{\textbf{Steg 2:}} \quad överkursövning visa steg 2 rigoröst.

\end{theorem}

\end{proof}
