\begin{theorem}[Prop 9.14]
\hfill \textcolor{magenta!80!black}{Bevislistan}

Låt $f, g$ holomorf i $D(z_0, R)$, $z_0$ enkelt \textcolor{magenta!80!black}{(ordning 1)} nollställe till $g$: Då är $\operatorname{Res}_{z_0} \dfrac{f}{g} = \dfrac{f(z_0)}{g'(z_0)}$
\end{theorem}
\begin{proof}
Enligt antagande $g(z) = (z - z_0)h(z)$, $h$ holomorf i $D(z_0, R)$, $h(z_0) = g'(z_0) \neq 0$. Får $h \neq 0$ i $D(z_0, \varepsilon)$, $\varepsilon > 0$.
\[ \Rightarrow \frac{f(z)}{g(z)} = \frac{f(z)/\rattat{h(z)}{h(z_0)}}{(z - z_0)}, \quad \frac{f}{h} \text{ holomorf i } D(z_0, \varepsilon) \]
\[ \overset{\textcolor{magenta!80!black}{\text{Prop 1.3R}}}{\Longrightarrow} \operatorname{Res}_{z_0} \frac{f}{g} = \frac{f(z_0)}{h(z_0)} = \frac{f(z_0)}{g'(z_0)} \]
\end{proof}

\begin{example}
Beräkna $\operatorname{Res}_\pi f$, $f(z) = \dfrac{e^z}{\sin z(z - 1)^2}$

\textbf{Lösning:} $\sin \pi = 0$, $\sin'(\pi) = \cos \pi = -1 \neq 0$
\[ \Big( g(z) = \sum c_k(z - z_0)^k = \underbrace{(z - z_0)^m}_{\textcolor{magenta!80!black}{\text{nollställe av ordning } m}} \sum_{k = 0}^\infty c_{k+m}(z - z_0)^k \Big) \]
$\Rightarrow \pi$ enkelt nollställe till $\sin z$.

$f(z) = \dfrac{g(z)}{\sin z}$, $g(z) = \dfrac{e^z}{(z - 1)^2}$ holomorf i omgivning till $\pi$
\[ \overset{\textcolor{magenta!80!black}{\text{Prop 9.14}}}{\Longrightarrow} \operatorname{Res}_\pi \frac{e^z/(z - 1)^2}{\sin z} = \frac{\left( e^z/(z - 1)^2 \right)_{z = \pi}}{(\sin z)'|_{z = \pi}} = \frac{e^\pi/(\pi - 1)^2}{\cos \pi} = -\frac{e^\pi}{(\pi - 1)^2} \]
\end{example}
