\fbox{8.26} \quad Hitta Laurentserien för $\sec z = \dfrac{1}{\cos z}$ runt $0$.

$\cos z = \displaystyle\sum_{n = 0}^\infty \frac{(-1)^n}{(2n)!} z^{2n}$ \qquad Ansats $\sec z = \displaystyle\sum_{m = 0}^\infty s_m z^m$
\begin{align*}
  1 &= \sec z \cdot \cos z = \sum_{n, m} \frac{(-1)^n s_m}{(2n)!} z^{2n + m} = \frac{(-1)^0 s_0}{(2 \cdot 0)!} + \frac{(-1)^0 s_1}{(2 \cdot 0)!} z + \left( \frac{(-1)^1 s_0}{(2 \cdot 1)!} + s_2 \right) z^2 + \\
  &+ \left( s_3 + \frac{(-1)^1}{(2 \cdot 1)!} s_1 \right) z^3 + \left( s_4 + \frac{(-1)^1}{(2 \cdot 1)!} s_2 + \frac{(-1)^2}{(2 \cdot 2)!} s_{\rattat{0}{6}} \right) z^4 = \\
  &\quad \textcolor{blue!60!black}{\left\{ \begin{array}{l} \text{alla olika sätt man kan få } z^4 \text{ med} \\ \text{olika } n \text{ och } m \quad (1, 0) \text{ eller } (2, 0) \end{array} \right\}} = \\
  &= \underset{1}{s_0} + \underset{0}{s_1} z + \underset{0}{\left( s_2 - \frac{s_0}{2} \right)} z^2 + \underset{0}{\left( s_3 - \frac{s_1}{2} \right)} z^3 + \underset{0}{\left( s_4 - \frac{s_2}{2} + \frac{s_0}{4!} \right)} z^4 + \dots
\end{align*}
\[ \left. \begin{array}{l} s_0 = 1 \\ s_1 = 0 \\ s_2 - \frac{s_0}{2} = 0 \\ s_3 - \frac{s_1}{2} = 0 \\ s_{\rattat{4}{1}} - \frac{s_2}{2} + \frac{s_0}{4!} = 0 \end{array} \right\} \quad \begin{array}{l} s_0 = 1 \\ s_1 = 0 \\ s_2 = 1/2 \\ s_3 = 0 \\ s_4 = \frac{1}{4} - \frac{1}{24} = \frac{5}{24} \end{array} \qquad \textcolor{blue!60!black}{\fbox{$\sec z = 1 + \dfrac{z^2}{2} + \dfrac{5}{24} z^4 + \dots$}} \]

\fbox{8.28} \quad Hitta multipliciteten av nollställerna för \quad a) $f(z) = e^z - 1$, $z_0 = 2\pi ik$ \quad b) $f(z) = \sin z - \tan z$, $z_0 = 0$

\textcircled{a} $f(2\pi ik) = e^{2\pi ik} - 1 = 0$, \ $f'(z) = e^z$ \quad $f'(2\pi ik) = 1 \neq 0$

\emph{multipliciteten} $= 1$

\textcircled{b} $f(0) = \sin(0) - \tan(0) = 0$
\[ f'(z) = \cos(z) - \frac{1}{\cos^2(z)}, \quad f'(0) = 1 - 1 = 0. \]
\[ f''(z) = -\sin z - \frac{2\cos z \sin z}{\cos^4 z} = -\sin(z) \rattat{-}{+} \frac{2\sin z}{\cos^3 z} \qquad f''(0) = 0 \]
\[ f'''(z) = \underbrace{-\cos z}_{-1} - \underset{2}{2} \underbrace{\frac{\cos^4 z - \sin z \cdot 3\cos^2 z(-\sin z)}{\cos^6 z}}_{1} = -3 \neq 0 \]
\emph{Multipliciteten} $= 3$
