\fbox{1.10} \quad Låt $a \in \mathbb{C}$, $b \in \mathbb{R}$. Visa att $|z|^2 + \operatorname{Re}(az) + b = 0$ är lösbar om.m $|a^2| \ge 4b$ och ekv.\ isåfall beskriver en cirkel.

\textbf{Lösning:} \emph{Ansats:} ekv för cirkel med radie $r$ centrerad i $w$ ges av $|z - w|^2 = r^2$

Får
\begin{align*}
  r^2 &= (z - w)\overline{(z - w)} = (z - w)(\bar{z} - \bar{w}) = |z|^2 + |w|^2 - z\bar{w} - \bar{z}w = \\
  &= |z^2| + |w|^2 - 2\operatorname{Re}(z\bar{w}) = |z|^2 + |w|^2 + \operatorname{Re}(z(-2\bar{w})) \qquad [\bar{z}w = \overline{z\bar{w}}]
\end{align*}
\[ \Leftrightarrow |z|^2 + \operatorname{Re}(z\underbrace{(-2\bar{w})}_{a}) + \underbrace{|w|^2 - r^2}_{b} = 0 \qquad r^2 \ge 0 \Leftrightarrow r^2 = |w|^2 - b = \underbrace{\frac{|a|^2}{4} - b}_{\ge 0} \]
\textcolor{blue!60!black}{$\Rightarrow$ Samma som i antagandet. $(|a|^2 \ge 4b)$}

\textcolor{blue!60!black}{(Blir då en cirkel med radie $\sqrt{\frac{|a^2|}{4} - b}$, centrerad i punkten $w = -\frac{\bar{a}}{2}$}

\textcolor{blue!60!black}{De är omm för att man kan alltid gå från $z, a, b$ till $z, w, r$ och tillbaka.}

\fbox{1.27f} \quad Beskriv mängden $\{z \in \mathbb{C} : |z| \ge \operatorname{Re} z + 1\}$

\textbf{Lösning:} $z = x + iy$, $|z| = \sqrt{x^2 + y^2}$, \ $|z| \ge \operatorname{Re} z + 1 \Leftrightarrow \sqrt{x^2 + y^2} \ge x + 1$
\notefigure{fig2}{}
Fall 1: $x \le -1$ \quad olikhet uppfylld

Fall 2: $x > -1$ \quad Både VL och HL $\ge 0$, så $\sqrt{x^2 + y^2} \ge x + 1 \Leftrightarrow$
\[ \Leftrightarrow x^2 + y^2 \ge x^2 + 2x + 1 \Leftrightarrow y^2 \ge 2x + 1 \]
\textcolor{blue!60!black}{Likhet gäller då $x = \dfrac{y^2 - 1}{2}$} \qquad \textcolor{blue!60!black}{\textbf{Svar:} allt till vänster om grafen $x = \dfrac{y^2 - 1}{2}$ + grafen}
