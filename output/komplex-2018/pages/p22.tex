\subsection{Homotopi mellan slutna kurvor}
\notefigure{fig2}{}
\begin{definition}
Kurvan $\gamma : [a, b] \to \mathbb{C}$ är \emph{sluten} om $\gamma(a) = \gamma(b)$
\end{definition}

Låt $\gamma_0, \gamma_1$ vara två slutna kurvor i område $G$.

\begin{definition}
$\gamma_0$ och $\gamma_1$ är \emph{homotopa i $G$}, skrivs $\gamma_0 \sim_G \gamma_1$, om $\exists$ kontinuerliga funktioner $\gamma(t, s) : [a, b] \times [0, 1] \to G$ s.a
\begin{enumerate}[label=\roman*)]
  \item $\gamma(t, 0) = \gamma_0(t)$
  \item $\gamma(t, 1) = \gamma_1(t)$
  \item $\gamma(a, s) = \gamma(b, s) \quad \forall s \in [0, 1]$
\end{enumerate}
dvs.\ kurvorna $\gamma_s(t) = \gamma(t, s)$ är alla slutna. Funktionen $\gamma(t, s)$ kallas \emph{homotopi} mellan $\gamma_0$ och $\gamma_1$
\end{definition}

\textbf{Intuitivt:} $\gamma_0 \sim_G \gamma_1$ om $\gamma_0$ kan kontinuerligt deformeras till $\gamma_1$ \emph{inne i $G$}
