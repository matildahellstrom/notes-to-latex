\begin{theorem}[Korollorium]
Om $f$ holomorf då existerar alla derivator $f^{(k)}$, $k \in \mathbb{N}$,
\[ f^{(k)}(w) = \frac{k!}{2\pi i} \int_{|z - w| = r} \frac{f(z)}{(z - w)^{k+1}}\,dz \qquad \textcolor{magenta!80!black}{\text{\underline{Tillhör inte bevislistan}}} \]
\end{theorem}
\begin{proof}
$f$ holomorf i $G$, $\overline{D}(w, r) \subseteq G$. CIF för $f'$ + CS $\Rightarrow$
\begin{align*}
  \Rightarrow \frac{f'(w + h) - f'(w)}{h} &= \frac{1}{2\pi i h} \int_{|z - w| = r} f(z)\left( \frac{1}{(z - w - h)^2} - \frac{1}{(z - w)^2} \right) dz = \\
  &= \left[ \frac{1}{(z - w - h)^2} - \frac{1}{(z - w)^2} = \frac{2h(z - w) - h^2}{(z - w - h)^2(z - w)^2} \right] = \\
  &= \frac{2}{2\pi i} \int_{|z - w| = r} \frac{f(z)}{(z - w - h)^2(z - w)}\,dz + O(h) \\
  &\xrightarrow[h \to 0]{} \frac{2}{2\pi i} \int_{|z - w| = r} \frac{f(z)}{(z - w)^3}\,dz \quad \text{som i förra beviset.}
\end{align*}
\end{proof}
