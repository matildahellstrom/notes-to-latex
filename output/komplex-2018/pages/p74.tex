\textbf{Repetition:}

\begin{theorem}[Residysatsen]
Antag $f$ holomorf i $G$ förutom isolerade $z_k$, $\gamma$ styckvis glatt sluten enkel positivt orienterad nollhomotop kurva i $G$ som undviker singulariteten $z_k$.
\notefigure{fig1}{}
Då är $\displaystyle\int_\gamma f\,dz = 2\pi i \sum_{z_k \in \operatorname{int}(\gamma)} \operatorname{Res}_{z_k} f$
\end{theorem}
\begin{itemize}
  \item $\operatorname{Res}_{z_0} f = c_{-1}$, \ $f(z) = \sum_{-\infty}^\infty c_k(z - z_0)^k$ i $D(z_0, R)^\times$
  \item Om $f(z) = \dfrac{g(z)}{(z - z_0)^m}$, $m \ge 1$, $g$ holomorf i omgivning $z_0$, då är $\operatorname{Res}_{z_0} f = \dfrac{g^{(m-1)}(z_0)}{(m - 1)!}$ \ \textcolor{blue!60!black}{(Prop 1.3 R)}
  \item Om $f, g$ holomorf i omgivning till $z_0$, $z_0$ enkelt nollställe till $g$, då är $\operatorname{Res}_{z_0} \dfrac{f}{g} = \dfrac{f(z_0)}{g'(z_0)}$ \ \textcolor{blue!60!black}{(Prop 9.14)}
\end{itemize}

\noindent\rule{\linewidth}{0.4pt}

\begin{example}
Beräkna $\displaystyle\int_{|z - 1| = 3} \frac{e^z}{\sin z(z - 1)^2}\,dz$

\textbf{Lösning:} $f(z) = \dfrac{e^z}{\sin z(z - 1)^2}$ holomorf i $\mathbb{C}$ förutom isol.\ sing.\ i $1$ samt $k\pi$, $k \in \mathbb{Z}$. $C(1, 3)$ uppfyller kraven:
\notefigure{fig2}{}
$0, 1, \pi$ ligger i $\operatorname{int}(C(1, 3))$
\[ \operatorname{Res}_1 f = \frac{e(\sin 1 - \cos 1)}{\sin^2 1} \qquad \operatorname{Res}_\pi f = \frac{e^\pi}{(\pi - 1)^2} \quad \textcolor{magenta!80!black}{\leftarrow \text{förra veckan}} \]
\[ \operatorname{Res}_0 f \underset{\textcolor{magenta!80!black}{\sin z \text{ enkelt nollställe i } 0}}{=} \left( \frac{e^z/(z - 1)^2}{\sin z} \right)_{z = 0} = 1 \]
\[ \int_\gamma f\,dz = 2\pi i\left( \frac{e(\sin 1 - \cos 1)}{\sin^2 1} - \frac{e^\pi}{(\pi - 1)^2} + 1 \right) \]
\end{example}
