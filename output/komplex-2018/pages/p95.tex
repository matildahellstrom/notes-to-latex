$\operatorname{Re} f(ti) = 0 \Leftrightarrow t = \pm 2$

$\operatorname{Im} f(ti) = 0 \Leftrightarrow \begin{cases} t = 0 \\ t = 1 \end{cases}$

Sida 8 \textcircled{4} Beräkna $I(t) = \displaystyle\int_{-\infty}^\infty \frac{1}{1 + s^2} \cdot \frac{1}{1 + (s - t)^2}\,ds$

\textbf{Lösning:} $f(s) = \dfrac{1}{1 + s^2}$ \quad $I(t) = f * f(t)$ enligt inversionsformeln.
\[ \hat{\hat{g}}(t) = 2\pi g(-t), \quad I(t) = f * f(t) = \frac{1}{2\pi} \hat{\widehat{f * f}}(-t) = \frac{1}{2\pi} \widehat{\hat{f} \cdot \hat{f}}(-t) \]
\[ = \frac{1}{2\pi} \widehat{(\pi e^{-|x|})^2}(-t) = \frac{\pi}{2} \widehat{e^{-2|x|}}(-t) = \frac{\pi}{2} \frac{4}{4 + t^2} = \frac{2\pi}{4 + t^2} \]

Sida 13 \quad Lös diff.ekvationen $\begin{cases} u'' + u = g(t) \\ u(0) = 0 \\ u'(0) = 0 \end{cases}$ \quad $g(t) = \begin{cases} 0 & 0 < t < \pi \\ 1 & t > \pi \end{cases}$
\notefigure{fig2}{}
\textbf{Lösning:}
\[ L(g) = L(u'' + u) = sL(u') - \cancel{u'(0)} + L(u) = s^2 L(u) \rattat{-}{+} s\cancel{u(0)} + L(u) = (s^2 + 1)L(u) \]
\[ L(u) = \frac{1}{s^2 + 1} L(g) = L(\sin x)L(g) = L(g * \sin t) \]
\[ \Rightarrow u(t) = g * \sin(t) = \int_0^t g(s)\sin(t - s)\,ds \overset{\textcolor{magenta!80!black}{t > \pi}}{=} \int_\pi^t \sin(t - s)\,ds = \]
\[ [\cos(t - s)]_\pi^t = 1 - \cos(t - \pi) \]
\[ u(t) = \begin{cases} 0 & t < \pi \\ 1 - \cos(t - \pi) & t > \pi \end{cases} \]
