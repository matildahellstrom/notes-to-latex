\fbox{9.1} \quad Anta att $f$ har ett nollställe av ordning $m$ vid $a$. Förklara varför $\frac{1}{f}$ har en pol vid ordning $m$ vid $a$.

\textbf{Lösning:} $f(z) = (z - a)^m g(z)$, $g(a) \neq 0$ nollställe av ordning $m$ vid $a$

$\to \dfrac{1}{f(z)} = (z - a)^{-m} \dfrac{1}{g(z)} \to \dfrac{1}{f}$ har en pol vid $a$ av ordning $m$.

\fbox{9.8} \quad \textcircled{d} Beräkna $\displaystyle\int_{C[0, 1]} \frac{dz}{z^2 \sin z}$

\textcircled{d} \quad $\sin z = z - \dfrac{z^3}{3!} + \dfrac{z^5}{5!} + \dots = zg(z)$ för någon funktion $g$.
\[ \operatorname{Res}_0\left( \frac{1}{z^2 \sin z} \right) = \frac{1}{2!} \lim_{z \to 0} \frac{d^2}{dz^2}\left( \frac{z^3}{z^2 \sin z} \right) \]
$\dfrac{1}{\sin z} = \dfrac{A}{z} + B + Cz + \dots$, \quad $1 = \sin z \cdot \dfrac{1}{\sin z} = \left( z - \dfrac{z^3}{3!} + \dfrac{z^5}{5!} - \dots \right)\left( \dfrac{A}{z} + B + Cz + \dots \right)$
\[ = \underbrace{(z) \cdot \left( \frac{A}{z} \right)}_{\textcolor{magenta!80!black}{= A}} \rattat{+}{-} Bz + z^2\left( C - \frac{A}{6} \right) + \dots \Rightarrow \begin{array}{l} A = 1 \\ B = 0 \\ C - \frac{A}{6} = 0 \Rightarrow C = \frac{1}{6} \end{array} \]
\[ \frac{1}{z^2 \sin z} = \frac{1}{z^2}\left( \frac{1}{z} + \frac{1}{6}z + \dots \right) = \frac{1}{z^3} + \frac{1}{6z} + \dots \quad \Big| \quad \int_{C[0, 1]} \frac{dz}{z^2 \sin z} = \frac{2\pi i}{6} = \underline{\frac{\pi i}{3}} \]
