\subsection{Cauchy-Riemanns ekvationer}
\begin{theorem}[(2.13)]
Låt $G \subseteq \mathbb{C}$ öppen, $f : G \to \mathbb{C}$, $u(x, y) = \operatorname{Re} f(x + iy)$, $v(x, y) = \operatorname{Im} f(x + iy)$ \ $(f = u + iv)$
\begin{enumerate}[label=\alph*)]
  \item Om $f$ holomorf i $G$ så uppfyller $u, v$ Cauchy-Riemanns ekvationer:
  \[ \begin{cases} \dfrac{du}{dx} = \dfrac{dv}{dy} \\[2ex] \dfrac{du}{dy} = -\dfrac{dv}{dx} \end{cases} \qquad \text{alt.} \quad \frac{df}{dx} = -i \frac{df}{dy} \]
  \item Om $f \in C^1$ (har kont.\ partiella derivator) och $u, v$ uppfyller CRs \textcolor{magenta!80!black}{(Cauchy-Riemanns)} ekvationer, då är $f$ holomorf.
\end{enumerate}
\end{theorem}

\begin{proof}[Bevis a)]
$f$ holomorf i $G$ enligt antagandet, så $f'(z_0)$ existerar $\forall z_0 \in G$.
\begin{align*}
  f'(z_0) &= \lim_{h \to 0} \frac{f(z_0 + h) \rattat{-}{+} f(z_0)}{h} = \textcolor{blue!60!black}{\left[\text{Låt } h = t + i0,\ z_0 = x_0 + iy_0\right]} = \\
  &= \lim_{t \to 0} \frac{f(x_0 + t, y_0) - f(x_0, y_0)}{t} = \rattat{\frac{du}{dx}}{\frac{df}{dx}}(x_0, y_0) + i \frac{dv}{dx}(x_0, y_0) = \\
  &= \textcolor{blue!60!black}{\left[\text{Låt nu } h = 0 + is,\ z_0 = x_0 + iy_0\right]} = \lim_{is \to 0} \frac{f(x_0, y_0 + s) - f(x_0, y_0)}{is} = \\
  &= -i \frac{df}{dy}(x_0, y_0) \quad \textcolor{magenta!80!black}{\left(\tfrac{1}{i} = -i\right)} \\
  &= -i \left( \frac{du}{dy}(x_0, y_0) + i \frac{dv}{dy}(x_0, y_0) \right) = \frac{dv}{dy}(x_0, y_0) - i \frac{du}{dy}(x_0, y_0)
\end{align*}
\[ \textcolor{blue!60!black}{\Rightarrow} \begin{cases} \dfrac{du}{dx} = \dfrac{dv}{dy} \ (= \operatorname{Re} f') \\[2ex] \dfrac{du}{dy} = -\dfrac{dv}{dx} \ (= -\operatorname{Im} f') \end{cases} \]
\end{proof}

\begin{example}
$g(z) = \bar{z} = x - iy$, \ $u(x, y) = x$, $v(x, y) = -y$

\textcolor{red!70!black}{$\Rightarrow$} $\dfrac{du}{dx} = 1$, $\dfrac{dv}{dy} = -1$ \textcolor{red!70!black}{$\Rightarrow$} $u, v$ uppfyller ej CRs ekvationer $\overset{\text{CR}}{\textcolor{red!70!black}{\Rightarrow}}$ $\bar{z}$ ej holomorf
\end{example}
