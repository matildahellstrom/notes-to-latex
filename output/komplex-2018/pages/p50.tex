\begin{theorem}[Kor 8.5]
Om $f(z) = \sum_0^\infty c_k(z - z_0)^k$, konvergensradie $R > 0$ då $c_k = \dfrac{f^{(k)}(z_0)}{k!}$
\end{theorem}
\begin{proof}
Iteration av Sats 8.2 ger $f^{(k)}(z) = \displaystyle\sum_{m \ge k} m(m - 1) \cdots (m - k + 1)c_m(z - z_0)^{m-k}$

$\Rightarrow f^{(k)}(z_0) = k!\,c_k$
\end{proof}

\begin{theorem}[Kor 8.6]
\hfill \textcolor{magenta!80!black}{Finns bara 1 potensserie till varje funktion (holomorf)}

Om $\sum_0^\infty c_k(z - z_0)^k = \sum_0^\infty d_k(z - z_0)^k$ i $D(z_0, r)$, $r > 0$, då är $c_k = d_k$, $\forall k \in \mathbb{N}$
\end{theorem}
\begin{proof}
$f(z) = \sum_0^\infty c_k(z - z_0)^k = \sum_0^\infty d_k(z - z_0)^k$
\[ c_k \underset{\textcolor{magenta!80!black}{8.5}}{=} \frac{f^{(k)}(z_0)}{k!} \underset{\textcolor{magenta!80!black}{8.5}}{=} d_k \]
\end{proof}
