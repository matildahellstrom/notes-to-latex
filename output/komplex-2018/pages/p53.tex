\fbox{7.25} \quad b) Hitta en potensserie \& avgör konvergensradien för $\dfrac{1}{3 - z/2}$. 

\hfill \fbox{$\dfrac{1}{1 - z} = \displaystyle\sum_{n = 0}^\infty z^n$ om $|z| < 1$}
\[ \frac{1}{3 - z/2} = \frac{1}{3}\left( \frac{1}{1 - z/6} \right) = \frac{1}{3} \sum_{n = 0}^\infty \left( \frac{z}{6} \right)^n \quad \left| \frac{z}{6} \right| < 1 \Leftrightarrow |z| < 6 \]

\fbox{7.26} \quad Hitta en potensserie runt origo för \ \fbox{a) $\cos z$ \quad b) $\cos(z^2)$ \quad c) $z^2 \sin z$ \quad d) $\sin^2(z)$}

\textcolor{blue!60!black}{alla funktioner är holomorfa. Går att utveckla alla som potensserie runt alla punkter.}

Vet att $f(z) = \displaystyle\sum_{n = 0}^\infty \frac{f^{(n)}(0)z^n}{n!} = \frac{f^{(2n)}(0)}{(2n)!} z^{2n}$

$\cos z = \displaystyle\sum_{n = 0}^\infty \frac{(-1)^n}{(2n)!} z^{2n}$ \quad \textcolor{magenta!80!black}{-- varierar mellan $+ \ 0 \ -$}

\textcircled{a}
\begin{tabular}{ll}
  $f(z) = \cos z$ & $f(0) = 1$ \\
  $f'(z) = -\sin z$ & $f'(0) = 0$ \\
  $f''(z) = -\cos z$ & $f''(0) = -1$ \\
  $f^{(3)}(z) = \sin z$ & $f^{(3)}(0) = 0$ \\
  $f^{(4)}(z) = \cos z$ & $= f(z)$
\end{tabular}

\textcircled{b} \textcolor{blue!60!black}{Hade kunnat göra samma, men vet vad $\cos z$ är}
\[ \cos(z^2) = \sum_0^\infty \frac{(-1)^n}{(2n)!} z^{4n} \]

\textcircled{c} Vet att $\sin z = -(\cos z)' = \rattat{-}{}\displaystyle\sum_{n = 1}^\infty \frac{(-1)^n}{(2n)!} 2n z^{2n-1} = \sum_{n = 1}^\infty \frac{(-1)^{n+1}}{(2n - 1)!} z^{2n-1}$

\textcolor{blue!60!black}{derivera serien term för term} \qquad $z^2 \sin z = \displaystyle\sum_{n = 1}^\infty \frac{(-1)^{n+1}}{(2n - 1)!} z^{2n+1}$ \ \textcolor{blue!60!black}{(multat med $z^2$)}
