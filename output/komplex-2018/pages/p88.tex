\begin{example}
Lös diff.ekv.
\[ (*) \quad -u''(t) + u(t) = g(t), \quad g \text{ given funktion i } L^1 \]
\textbf{Lösning:} Vi antar $u, u' \in L^1$. Får $u'' \in L^1$

FT av $(*)$ ger: $\hat{g} = -\widehat{u''} + \hat{u} = (-ix)^2\hat{u} + \hat{u} = (1 + x^2)\hat{u} \Rightarrow$
\[ \Rightarrow \hat{u} = \rattat{\hat{g}}{g} \cdot \frac{1}{1 + x^2} = \left[ \widehat{e^{-|t|}} = \frac{2}{1 + x^2} \right] = \frac{1}{2}\hat{g}\,\widehat{e^{-|t|}} = \widehat{\tfrac{1}{2} g * e^{-|t|}} \]
\[ \Longrightarrow u = \frac{1}{2} g * e^{-|t|}, \text{ dvs.\ } u(t) = \frac{1}{2} \int_{-\infty}^\infty g(r)e^{-|t - r|}\,dr. \]
\textcolor{red!70!black}{\textbf{OBS!}} Hittar endast lösning i $L^1$ allmänna lösningen: $\frac{1}{2} g * e^{-|t|} + C_1 e^t + C_2 e^{-t}$
\end{example}

\subsection{Laplacetransform}
\begin{definition}
Låt $f : [0, \infty) \to \mathbb{C}$, integrerbar, och anta $\exists A, B \in \mathbb{R}$ s.a.\ $|f(t)| \le Ae^{Bt}$, då definierar vi Laplacetransformen (LF) av $f$:
\notefigure{fig1}{}
\[ \tilde{f}(s) = \int_0^\infty f(t)e^{-st}\,dt, \quad s \in \mathbb{C} \text{ s.a.\ } \operatorname{Re} s > B \]
\end{definition}

\textcolor{red!70!black}{\textbf{OBS!}} $\tilde{f}$ är holomorf i $\{s : \operatorname{Re} s > B\}$,
\[ \widetilde{f}'(s) = \int_0^\infty f(t)(e^{-st})'\,dt = -\int_0^\infty tf(t)e^{-st}\,dt = -\widetilde{tf}(s) \]

\noindent\rule{\linewidth}{0.4pt}

\begin{theorem}[Inversionsformel för LT (Sats 3.27)]
Om $|f(t)| \le Ae^{Bt}$, $C > B$ och $\displaystyle\int_{-\infty}^\infty |\tilde{f}(c + iy)|\,dy < \infty$ då är
\[ f(t) = \frac{1}{2\pi} \int_{\operatorname{Re} s = c} \tilde{f}(s)e^{st}\,ds \]
\notefigure{fig2}{}
\emph{Speciellt}: $\tilde{f} = \tilde{g} \Rightarrow f = g$
\end{theorem}
