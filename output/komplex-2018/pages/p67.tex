\fbox{7.34} \quad Hitta funktionen som ges av potensserien \quad \textcircled{b} $\displaystyle\sum_1^\infty k(z - 1)^{k-1}$ \quad \textcircled{c} $\displaystyle\sum_2^\infty k(k - 1)z^k$

\textbf{Lösning:} \textcircled{b}
\[ \sum_1^\infty k(z - 1)^{k-1} \underset{\textcolor{magenta!80!black}{\text{Sats}}}{=} \left( \sum_0^\infty (z - 1)^k \right)' \overset{|z - 1| < 1}{=} \left( \frac{1}{1 - (z - 1)} \right)' = \left( \frac{1}{2 - z} \right)' = \frac{1}{(2 - z)^2} \]
\textcircled{c}
\[ \sum_2^\infty k(k - 1)z^k = z^2 \sum_2^\infty k(k - 1)z^{k-2} \overset{\textcolor{magenta!80!black}{\text{Sats}}}{=} z^2\left( \sum_0^\infty z^k \right)'' \overset{|z| < 1}{=} z^2\left( \frac{1}{1 - z} \right)'' = \]
\[ = z^2\left( \frac{1}{(1 - z)^2} \right)' = z^2 \frac{2}{(1 - z)^3} = \frac{2z^2}{(1 - z)^3} \]

\fbox{7.35} \quad Låt $f(z) = \displaystyle\int_{[0, 1]} \frac{dw}{1 - wz}$, $z \in D(0, 1)$. Hitta en potensserie för $f$ centrerad i $0$.

\textbf{Lösning:} $|wz| \le |z| < 1$ på $[0, 1] \Rightarrow \dfrac{1}{1 - wz} = \displaystyle\sum_0^\infty (wz)^k$ konvergerar likformigt på $[0, 1]$
\[ \Rightarrow f(z) = \int_{[0, 1]} \sum_0^\infty (wz)^k\,dw \overset{\textcolor{magenta!80!black}{\text{likf konv}}}{=} \sum_0^\infty z^k \int_{[0, 1]} w^k\,dw = \]
\[ = \sum_0^\infty z^k \left[ \frac{w^{k+1}}{k + 1} \right]_0^1 = \sum_0^\infty z^k \cdot \frac{1}{k + 1} \qquad \left( \rattat{\sum_0^\infty \frac{z^k}{k + 1} = -\frac{\operatorname{Log}(1 - z)}{z}}{\sum_1^\infty \frac{z^k}{k} = \frac{\operatorname{Log}(1 - z)}{z}} \right) \]

\fbox{8.5} \quad Bestäm 3 första termerna i Taylorutvecklingen av: \quad \textcircled{a} $\dfrac{1}{1 + z^2}$ centrerad i $1$ \quad \textcircled{d} $e^{z^2}$ centrerad i $i$

\textbf{Lösning:} \textcircled{a} Sätt $w = z - 1$, $z = w + 1$, $z^2 = w^2 + 2w + 1$
\[ \frac{1}{1 + z^2} = \frac{1}{2 + 2w + w^2} = \frac{1}{2} \frac{1}{1 - (-w - \frac{w^2}{2})} \overset{|w| \ll 1}{=} \frac{1}{2} \sum_0^\infty (-1)^k\left( w + \frac{w^2}{2} \right)^k = \]
\[ = \frac{1}{2}\left( 1 - \left( w + \frac{w^2}{2} \right) + \left( w + \frac{w^2}{2} \right)^2 + O(w^3) \right) = \frac{1}{2} - \frac{1}{2}w - \frac{1}{4}w^2 \textcolor{magenta!80!black}{\left( + \frac{1}{2}w^2 \right)} + O(w^3) \]
\[ = \frac{1}{2} - \frac{1}{2}(z - 1) + \rattat{\frac{1}{4}}{\frac{1}{2}}(z - 1)^2 + O((z - 1)^3) \]
\textcircled{d} $w = z - i$, $z = w + i$, $z^2 = w^2 + 2iw - 1$
\[ e^{z^2} = e^{w^2 + \rattat{2iw}{2w} - 1} = e^{-1}e^{2iw}e^{w^2} = e^{-1}\left( 1 + 2iw + \frac{(2iw)^2}{2} + O(w^3) \right) \times \]
\[ \times (1 + w^2 + O(w^4)) = \frac{1}{e}(1 + 2iw - w^2) + O(w^3) = \frac{1}{e}(1 + 2i(z - i) - (z - i)^2) + O((z - i)^3) \]
