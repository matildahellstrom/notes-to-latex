\begin{example}
Bestäm antalet nollställen (med mult) till $z^5 + 2z + 1$ i högra halvplanet

\textbf{Lösning:}
\notefigure{fig1}{}
\[ \gamma_R(t) = Re^{it},\ t \in [-\pi/2, \pi/2] \qquad \sigma_R = \gamma_R \cup [iR, -iR] \]
$f(z) = z^5 + 2z + 1$, \quad $W(f \circ \sigma_R) = ?$

\emph{$\operatorname{argvar}(f \circ \gamma_R)$}:
\[ f(Re^{it}) = (Re^{it})^5 + 2Re^{it} + 1 = R^5 e^{5it}\Big( 1 + \underbrace{\frac{2e^{-4it}}{R^4} + \frac{e^{-5it}}{R^5}}_{\textcolor{magenta!80!black}{\text{litet},\ R \gg 1}} \Big) \]
\[ \Rightarrow \arg(f(Re^{it})) \approx \arg(R^5 e^{5it}) = 5t + 2\pi k,\ R \gg 1 \]
$\Rightarrow \operatorname{argvar}(f \circ \gamma_R) \approx 5\pi$
\notefigure{fig2}{}

\emph{$\operatorname{argvar}(f \circ [iR, -iR])$}:
\notefigure{fig3}{}
\[ f(iy) = iy^5 + 2iy + 1 = 1 + i(y^5 + 2y) \]
\[ f(iR) = 1 + i(R^5 + 2R) \Rightarrow \operatorname{Arg}(f(iR)) \approx \pi/2 \]
\[ f(-iR) = 1 + i(-R^5 - 2R) \Rightarrow \operatorname{Arg}(f(\rattat{-iR}{-R})) \approx -\pi/2 \]
$\operatorname{Re}(f(iy)) = 1 > 0 \Rightarrow f \circ [iR, -iR]$ passerar noll på höger sida
\begin{align*}
  &\Rightarrow \operatorname{argvar}(f \circ [iR, -iR]) \approx -\pi \\
  &\Rightarrow \operatorname{argvar}(f \circ \sigma_R) \approx 5\pi - \pi = 4\pi \\
  &\Rightarrow \operatorname{argvar}(f \circ \sigma_R) = 4\pi,\ R \gg 1 \\
  &\Rightarrow W(f \circ \sigma_R) = 2,\ R \gg 1
\end{align*}
$\overset{\textcolor{magenta!80!black}{\text{arg princ.}}}{\Longrightarrow} f$ har 2 nollställen i $\operatorname{int}(\sigma_R)$, $R \gg 1$, dvs 2 nollställen i högra halvplanet
\end{example}
