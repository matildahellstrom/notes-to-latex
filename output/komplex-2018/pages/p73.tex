\fbox{9.6} \quad Antag $f$ har enkel pol (ord.\ 1) i $z_0$, $g$ holomorf i omgivning till $z_0$. Visa att $\operatorname{Res}_{z_0} fg = g(z_0) \operatorname{Res}_{z_0} f$

\textbf{Lösning:} $f$ enkel pol betyder $f(z) = \dfrac{h(z)}{z - z_0}$, $h$ holomorf, $h(z_0) \neq 0$. \ $\operatorname{Res}_{z_0} f = h(z_0)$
\[ \operatorname{Res}_{z_0} fg = \operatorname{Res}_{z_0}\left( \frac{g(z)h(z)}{z - z_0} \right) \overset{\text{räkneregel}}{=} g(z_0)h(z_0) = g(z_0) \operatorname{Res}_{z_0} f \]

\fbox{9.7} \quad \textcircled{e} Bestäm $\operatorname{Res}_0\left( \dfrac{e^{4z} - 1}{\sin^2 z} \right)$

\textbf{Lösning:} $\sin z = z - \dfrac{z^3}{6} + O(z^5)$

$e^{4z} = 1 + 4z + O(z^2)$, \quad $\dfrac{1}{\sin z} = \{\text{se uppgift 8.19}\} = \dfrac{1}{z} + O(z)$

$\dfrac{1}{\sin^2 z} = \dfrac{1}{z^2} + O(1)$
\[ \frac{e^{4z} - 1}{\sin^2 z} = (4z + O(z^2))\left( \frac{1}{z^2} + O(1) \right) = \textcolor{magenta!80!black}{\underline{\textcolor{black}{\frac{4}{z} + O(1)}}} \Rightarrow \operatorname{Res}_0\left( \frac{e^{4z} - 1}{\sin^2 z} \right) = 4 \]

\fbox{9.8} \quad \textcircled{c} Beräkna $\displaystyle\int_{|z| = 2} \frac{e^z}{z^3 + z}\,dz$

\textbf{Lösning:} $z^3 + z = z(z + i)(z - i)$, enkla nollställen i $0, \pm i$
