\subsubsection*{Linjeintegral kring en sluten kurva}
\begin{example}
\[ \vb{F} = \begin{bmatrix} y \\ x \\ z \end{bmatrix}, \quad \text{Parametrisering} = \begin{bmatrix} x = \cos\theta \\ y = \sin\theta \\ z = 0 \end{bmatrix}, \quad \theta \in [0, 2\pi] \]
\[ \frac{d\vb{r}}{d\theta} = \begin{bmatrix} -\sin\theta \\ \cos\theta \\ 0 \end{bmatrix}, \quad \vb{F}(\theta) = \begin{bmatrix} \sin\theta \\ \cos\theta \\ 0 \end{bmatrix} \]
\begin{align*}
\oint_C \vb{F} \cdot d\vb{r} &= \int_0^{2\pi} (\sin\theta, \cos\theta, 0) \cdot (-\sin\theta, \cos\theta, 0)\,d\theta = \\
&= \int_0^{2\pi} -\sin^2\theta + \cos^2\theta\,d\theta = \int_0^{2\pi} \cos 2\theta\,d\theta = \left[ \frac{1}{2}\sin 2\theta \right]_0^{2\pi} = 0
\end{align*}
Fältet $\vb{F}$ är konservativt.
\end{example}
